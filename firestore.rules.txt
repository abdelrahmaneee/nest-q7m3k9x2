rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /fam/{fam}/d/{doc} {
      allow get: if request.auth != null && fam.size() == 14;
      allow create, update: if request.auth != null
        && fam.size() == 14
        && doc in ['kid', 'coach']
        && request.resource.data.keys().hasOnly(['j', 'u'])
        && request.resource.data.j is string
        && request.resource.data.j.size() < 900000;
      allow list, delete: if false;
    }
  }
}
